/* ========================= OCR 強化版 ========================= */ async function runOCR(file){ if(!window.Tesseract){ await loadTesseract(); } /* 画像をOCR用に加工 ・拡大 ・コントラスト調整 ・明るさ調整 */ const processedFile = await preprocessImage(file); /* OCR① 通常モード */ const result1 = await Tesseract.recognize( processedFile, "eng", { logger:m => { if( m.status === "recognizing text" ){ const progress = Math.round( (m.progress || 0) * 100 ); document .getElementById("photoInfo") .textContent = `📖 写真を読み取り中... ${progress}%`; } }, config:{ tessedit_pageseg_mode:"6" } } ); /* OCR② バラバラの文字を探すモード */ const result2 = await Tesseract.recognize( processedFile, "eng", { config:{ tessedit_pageseg_mode:"11" } } ); /* 2つのOCR結果を合体 */ const text1 = result1?.data?.text || ""; const text2 = result2?.data?.text || ""; console.log( "OCR①:", text1 ); console.log( "OCR②:", text2 ); return ( text1 + "\n" + text2 ); } /* ========================= OCR用画像加工 ========================= */ function preprocessImage(file){ return new Promise( (resolve,reject) => { const img = new Image(); const reader = new FileReader(); reader.onload = () => { img.onload = () => { /* 元画像を拡大 */ const scale = Math.max( 2, 1800 / img.width ); const canvas = document.createElement( "canvas" ); canvas.width = Math.round( img.width * scale ); canvas.height = Math.round( img.height * scale ); const ctx = canvas.getContext( "2d" ); /* OCRしやすくする */ ctx.filter = "contrast(1.35) brightness(1.08)"; ctx.drawImage( img, 0, 0, canvas.width, canvas.height ); canvas.toBlob( blob => { if(!blob){ reject( new Error( "OCR用画像の作成に失敗しました" ) ); return; } resolve( new File( [blob], "ocr-image.jpg", { type: "image/jpeg" } ) ); }, "image/jpeg", 0.95 ); }; img.onerror = () => { reject( new Error( "画像の読み込みに失敗しました" ) ); }; img.src = reader.result; }; reader.onerror = () => { reject( new Error( "画像ファイルの読み込みに失敗しました" ) ); }; reader.readAsDataURL(file); } ); } /* ========================= Tesseract読み込み ========================= */ function loadTesseract(){ return new Promise( (resolve,reject) => { /* すでに読み込まれている場合 */ if(window.Tesseract){ resolve(); return; } const script = document.createElement( "script" ); script.src = "https://cdn.jsdelivr.net/npm/tesseract.js@5/dist/tesseract.min.js"; script.onload = resolve; script.onerror = () => { reject( new Error( "OCRエンジンの読み込みに失敗しました" ) ); }; document .head .appendChild(script); } ); } /* ========================= OCR解析 ========================= */ function parseTournamentText(text){ const result = { tournamentName:"", players:"", buyIn:"" }; const clean = String(text || "") .replace(/\r/g," ") .replace(/[|]/g," ") .replace(/\s+/g," ") .trim(); console.log( "OCR全文:", clean ); /* ========================= 参加人数 ========================= 例： PLAYERS 2/26 → 26 */ let playerMatch = clean.match( /PLAYERS?\s*[:\-]?\s*\d+\s*\/\s*(\d+)/i ); /* OCRで PLAYERS が認識できなかった場合 2/26 のような表記を探す */ if(!playerMatch){ playerMatch = clean.match( /\b\d+\s*\/\s*(\d+)\b/ ); } if(playerMatch){ result.players = playerMatch[1]; } /* ========================= Buy-in ========================= 例： ENTRY 3.5K ENTRY ¥3,500 BUY-IN 3.5K */ let buyInMatch = clean.match( /(?:BUY[\s-]*IN|ENTRY\s*FEE|ENTRY)\s*[:\-]?\s*[¥￥]?\s*([\d,]+(?:\.\d+)?\s*K?)/i ); if(buyInMatch){ result.buyIn = normalizeMoney( buyInMatch[1] ); } /* 「3.5K」だけを認識した場合 */ if(!result.buyIn){ const kMatches = clean.match( /\b\d+(?:\.\d+)?\s*K\b/gi ); if(kMatches){ /* 最初に見つかったK表記 */ result.buyIn = normalizeMoney( kMatches[0] ); } } /* ========================= 大会名 ========================= */ /* OCR結果を行単位でも確認 */ const lines = String(text || "") .replace(/\r/g,"") .split("\n") .map( line => line .replace(/\s+/g," ") .trim() ) .filter(Boolean); console.log( "OCR行:", lines ); /* 大会名候補 */ const candidates = lines .filter( line => line.length >= 3 ) .filter( line => !/^(LEVEL|PLAYERS?|AVERAGE|CHIPCOUNT|INTERVAL|GAME TIME|ENTRY|ENTRY FEE|REENTRY|RE-ENTRY|BIG-ANTE|REGISTRATION|BLINDS?|ANTE)$/i .test(line) ) .filter( line => !/^\d+\s*\/\s*\d+$/.test( line ) ) .filter( line => !/^\d{1,2}:\d{2}$/.test( line ) ); /* 大会名っぽい文字を優先 */ const tournamentCandidate = candidates.find( line => /NLH|PLO|HOLIDAY|MAIN|EVENT|TOURNAMENT|DAILY|TURBO|DEEP|MYSTERY|CHAMPIONSHIP|POKER/i .test(line) ); if(tournamentCandidate){ result.tournamentName = cleanTournamentName( tournamentCandidate ); }else if( candidates.length ){ result.tournamentName = cleanTournamentName( candidates[0] ); } console.log( "OCR解析結果:", result ); return result; } /* ========================= 大会名クリーニング ========================= */ function cleanTournamentName(name){ return String( name || "" ) .replace( /^[^a-zA-Z0-9ぁ-んァ-ヶ一-龯]+/, "" ) .replace( /[^a-zA-Z0-9ぁ-んァ-ヶ一-龯\s&'’\-#]+$/g, "" ) .replace( /\s+/g, " " ) .trim(); } /* ========================= 金額変換 ========================= */ function normalizeMoney(value){ if(!value){ return 0; } const text = String(value) .replace(/,/g,"") .replace(/円/g,"") .replace(/¥/g,"") .replace(/￥/g,"") .trim() .toUpperCase(); /* 3.5K ↓ 3500 */ if( text.endsWith("K") ){ const number = Number( text.replace( "K", "" ) ); if( !Number.isNaN(number) ){ return Math.round( number * 1000 ); } } /* 普通の数字 */ const number = Number(text); return Number.isNaN(number) ? 0 : number; }
